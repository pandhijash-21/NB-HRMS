2. Production Nginx Reverse Proxy Configuration

To support SSE without buffering and WebSocket upgrades, use the following production Nginx block: 

# /etc/nginx/sites-available/nb-platform.conf
server {
    listen 443 ssl http2;
    server_name crm.nbsince1946.com;

    ssl_certificate /etc/letsencrypt/live/crm.nbsince1946.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/crm.nbsince1946.com/privkey.pem;

    # Core REST API
    location /api/ {
        proxy_pass http://127.0.0.1:4000/api/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Real-Time Server-Sent Events (SSE) - Crucial for zero buffering
    location /api/events/stream {
        proxy_pass http://127.0.0.1:4000/api/events/stream;
        proxy_http_version 1.1;
        proxy_set_header Connection '';
        proxy_buffering off;
        proxy_cache off;
        chunked_transfer_encoding off;
        proxy_read_timeout 24h;
    }

    # Socket.IO WebSocket Gateway
    location /socket.io/ {
        proxy_pass http://127.0.0.1:4000/socket.io/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_read_timeout 86400;
    }
}
